cmake_minimum_required(VERSION 3.20)
project(baro_consumer C)
set(CMAKE_C_STANDARD 99)
if(BARO_SOURCE)
    set(BARO_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    add_subdirectory("${BARO_SOURCE}" baro)
else()
    # Requesting the version exercises the installed package version file.
    find_package(baro ${BARO_VERSION} CONFIG REQUIRED)
endif()
enable_testing()
if(NOT MSVC)
    add_compile_options(-Wall -Wextra -Wpedantic -Werror)
endif()
add_library(test_archive STATIC archive.c)
target_link_libraries(test_archive PRIVATE baro::baro)
add_executable(default_runner test.c)
target_link_libraries(default_runner PRIVATE test_archive baro::main)
add_test(NAME default_runner COMMAND default_runner)
add_executable(custom_runner custom.c test.c)
target_link_libraries(custom_runner PRIVATE test_archive baro::baro)
add_test(NAME custom_runner COMMAND custom_runner)
add_test(NAME custom_isolated COMMAND custom_runner --isolate)

baro_discover_tests(default_runner TEST_PREFIX "discovered::" LABELS integration TIMEOUT 5)

baro_discover_tests(custom_runner TEST_PREFIX "" TEST_SUFFIX "::custom" TIMEOUT 5)
